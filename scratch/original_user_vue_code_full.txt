function toggleCategory(categoryName: string) {
  expandedCategory.value =
    expandedCategory.value === categoryName ? null : categoryName
}

function selectMainCategory(categoryName: string) {
  selectedCategory.value = categoryName
  selectedSubcategory.value = ''
}

function selectSubcategory(categoryName: string, subcategory: string) {
  selectedCategory.value = categoryName
  selectedSubcategory.value = subcategory
}

function toggleFilterSection(section: string) {
  expandedFilterSection.value =
    expandedFilterSection.value === section ? null : section
}

function clearFilters() {
  searchText.value = ''
  selectedCity.value = ''
  selectedCategory.value = ''
  selectedSubcategory.value = ''
  selectedSector.value = ''
  selectedType.value = ''
  selectedMethod.value = ''
  selectedPricingType.value = ''
}

const filteredTenders = computed(() => {
  return tenders.value.filter((tender) => {
    const searchValue = searchText.value.toLocaleLowerCase('tr')

    const searchMatch =
      !searchText.value ||
      tender.title.toLocaleLowerCase('tr').includes(searchValue) ||
      tender.company.toLocaleLowerCase('tr').includes(searchValue) ||
      tender.description.toLocaleLowerCase('tr').includes(searchValue)

    const cityMatch =
      !selectedCity.value ||
      tender.city === selectedCity.value

    const categoryMatch =
      !selectedCategory.value ||
      tender.category === selectedCategory.value

    const sectorMatch =
      !selectedSector.value ||
      tender.sector === selectedSector.value

    const typeMatch =
      !selectedType.value ||
      tender.type === selectedType.value

    const methodMatch =
      !selectedMethod.value ||
      tender.method === selectedMethod.value

    const pricingMatch =
      !selectedPricingType.value ||
      tender.pricing === selectedPricingType.value

    return (
      searchMatch &&
      cityMatch &&
      categoryMatch &&
      sectorMatch &&
      typeMatch &&
      methodMatch &&
      pricingMatch
    )
  })
})
